public class Main {

    static class CinemaBooking {
        String customerName;
        String movieName;
        double ticketPrice;
        int numberOfTickets;

        public CinemaBooking(String customerName, String movieName, double ticketPrice, int numberOfTickets) {
            this.customerName = customerName;
            this.movieName = movieName;
            this.ticketPrice = ticketPrice;
            this.numberOfTickets = numberOfTickets;
        }

     
        public CinemaBooking() {
            this.customerName = "";
            this.movieName = "";
            this.ticketPrice = 0.0;
            this.numberOfTickets = 0;
        }

        
        public void bookTickets(String customerName, String movieName, double ticketPrice, int numberOfTickets) {
            this.customerName = customerName;
            this.movieName = movieName;
            this.ticketPrice = ticketPrice;
            this.numberOfTickets = numberOfTickets;
        }

        public void displayBooking() {
            System.out.println("Customer Name     : " + this.customerName);
            System.out.println("Movie Name        : " + this.movieName);
            System.out.println("Ticket Price      : " + this.ticketPrice);
            System.out.println("Number of Tickets : " + this.numberOfTickets);
            System.out.println("Total Amount      : " + (this.ticketPrice * this.numberOfTickets));
            System.out.println("---------------------------------------------");
        }

        
        public boolean compareCustomer(CinemaBooking other) {
            return this.customerName.equalsIgnoreCase(other.customerName);
        }
    }

    public static void main(String[] args) {
      
        CinemaBooking booking1 = new CinemaBooking("Ram", "Avatar", 200.0, 5);
        booking1.displayBooking();

        
        CinemaBooking booking2 = new CinemaBooking();
        booking2.bookTickets("Ravana", "Avatar", 200.0, 2);
        booking2.displayBooking();

      
        System.out.println("Customer Name Comparison:");
        if (booking1.compareCustomer(booking2)) {
            System.out.println("Both bookings belong to the same customer.");
        } else {
            System.out.println("Bookings belong to different customers.");
        }
    }
}
